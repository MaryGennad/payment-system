Интеграция рекуррентных платежей ЮKassa

Привет! Я подготовила для тебя подробную инструкцию по подключению привязки карт и автоматического списания 1 рубля при успешной проверке марки. Всё протестировано на моём проекте, по шагам — должно заработать.

Что нужно сделать: Добавить в твой проект возможность один раз привязать карту пользователя, а затем автоматически списывать 1 рубль при каждой успешной проверке марки в "Честном ЗНАКе".

Общая схема работы

1. Фронтенд: Запрос на привязку карты
→
2. Бэкенд: Создание платежа в ЮKassa
(save_payment_method: true, metadata: userId)
→
3. ЮKassa: Вебхук "payment.succeeded"
→
4. Бэкенд: Сохранение payment_method_id (токена) в БД
5. Фронтенд: Сканирование марки → Запрос на списание 1₽
→
6. Бэкенд: Создание платежа через payment_method_id (capture: true)

Предварительные требования

0
Если ЮKassa ещё не подключена

Перед тем как начать, нужно зарегистрироваться в ЮKassa и получить доступ к API:

  1. Зарегистрируйся на yookassa.ru
  2. Создай магазин (если ещё нет)
  3. Перейди в раздел "Интеграция" → "Ключи API"
  4. Скопируй Shop ID и Secret Key
  5. Эти данные нужно будет добавить в конфиг проекта (см. Этап 2)

Эндпоинт для привязки карты

Добавь этот роут после app.delete("/api/deleteUser/:userId", ...) (примерно строка 305), перед комментарием //MARK INFO ---->.

server.js
app.post("/api/save-card", [checkToken], async (req, res) => {
  try {
    const { email, amount = 1, description = 'Привязка карты для будущих платежей' } = req.body;

    const yookassaResponse = await axios.post(
      'https://api.yookassa.ru/v3/payments',
      {
        amount: { value: parseFloat(amount).toFixed(2), currency: 'RUB' },
        confirmation: {
          type: 'redirect',
          return_url: `${exports.FRONTEND_URL}/cards.html?status=card-saved`
        },
        capture: true,
        save_payment_method: true,
        description: description,
        payment_method_data: { type: 'bank_card' },
        metadata: {
          userId: req.user._id.toString()
        },
        receipt: {
          customer: { email: email },
          items: [{
            description: description,
            quantity: '1.00',
            amount: { value: parseFloat(amount).toFixed(2), currency: 'RUB' },
            vat_code: 2
          }]
        }
      },
      {
        auth: {
          username: exports.yookassa.shopId,
          password: exports.yookassa.secretKey
        },
        headers: {
          'Content-Type': 'application/json',
          'Idempotence-Key': `save-card-${req.user._id}-${Date.now()}`
        }
      }
    );

    res.json({
      success: true,
      confirmation_url: yookassaResponse.data.confirmation.confirmation_url
    });
  } catch (err) {
    console.error('Save card error:', err.response?.data || err.message);
    res.status(500).json({
      error: err.response?.data?.description || 'Ошибка привязки карты'
    });
  }
});

Вебхук для обработки уведомлений от ЮKassa

Добавь этот роут сразу после предыдущего. Он будет принимать уведомления от ЮKassa и сохранять токены карт.

server.js
app.post("/api/payments/webhook", async (req, res) => {
  try {
    const event = req.body;

    if (event.event === 'payment.succeeded') {
      const paymentData = event.object;
      const paymentMethodId = paymentData.payment_method?.id;
      const userId = paymentData.metadata?.userId;

      if (paymentData.payment_method?.saved && 
          parseFloat(paymentData.amount.value) === 1 && 
          userId) {

        const cardData = paymentData.payment_method.card;

        if (cardData) {
          const existingCard = await require('./models/card.model').findOne({
            user: userId,
            last4: cardData.last4
          });

          if (!existingCard) {
            await new require('./models/card.model')({
              user: userId,
              cardToken: paymentMethodId,
              last4: cardData.last4,
              cardType: cardData.card_type,
              isActive: true
            }).save();

            console.log(`[YOOKASSA] Токен карты ****${cardData.last4} сохранен для пользователя ${userId}`);
          }
        }
      }
    }

    res.status(200).json({ status: 'ok' });
  } catch (err) {
    console.error('Webhook processing error:', err);
    res.status(200).json({ status: 'ok' });
  }
});

Функция для рекуррентного списания

Добавь эту функцию перед app.post("/api/getMarkInfo", ...) (примерно строка 318).

server.js
async function chargeForMarkCheck(userId, markCode) {
  try {
    const CardModel = require('./models/card.model');
    
    const card = await CardModel.findOne({ 
      user: userId, 
      isActive: true 
    });

    if (!card || !card.cardToken) {
      console.log(`[YOOKASSA] Карта не найдена для пользователя ${userId}. Списание пропущено.`);
      return { success: false, reason: 'no_card' };
    }

    const response = await axios.post(
      'https://api.yookassa.ru/v3/payments',
      {
        amount: { value: '1.00', currency: 'RUB' },
        capture: true,
        description: `Проверка марки: ${markCode}`,
        payment_method_id: card.cardToken
      },
      {
        auth: {
          username: exports.yookassa.shopId,
          password: exports.yookassa.secretKey
        },
        headers: {
          'Content-Type': 'application/json',
          'Idempotence-Key': `mark-check-${userId}-${Date.now()}`
        }
      }
    );

    console.log(`[YOOKASSA] Успешное списание 1 RUB. Mark: ${markCode}, Payment ID: ${response.data.id}`);
    return { 
      success: true, 
      paymentId: response.data.id, 
      status: response.data.status 
    };

  } catch (err) {
    console.error(`[YOOKASSA] Ошибка списания за марку ${markCode}:`, err.response?.data || err.message);
    return { success: false, error: err.message };
  }
}

Настройка вебхука в личном кабинете ЮKassa

Это важный шаг! Без него токены не будут сохраняться.

  1. Зайди в ЮKassa → Настройки → HTTP-уведомления
  2. Нажми "Добавить URL"
  3. Введи адрес: https://твой-домен.ru/api/payments/webhook
  4. Выбери событие: payment.succeeded
  5. Нажми "Сохранить"
9
Включение автоприема платежей

Чтобы платежи не зависали в статусе "Ожидает подтверждения":

  1. Зайди в ЮKassa → Настройки → Способы оплаты → Банковская карта
  2. Включи опцию "Автоприем платежей"

Важное архитектурное ограничение

Токенизация карт работает на уровне процессинга ЮKassa. В мобильных приложениях банков (Сбер, Т-Банк и др.) пользователь не увидит активной "подписки" в привычном понимании, так как для банка это серия независимых инициативных транзакций мерчанта (MIT). Управление привязкой (добавление/удаление карты) должно быть реализовано в интерфейсе твоего веб-приложения путем создания или удаления записей в модели Card.

Как проверить, что всё работает

  1. Вызови POST /api/save-card с валидным токеном авторизации
  2. Перейди по полученной confirmation_url
  3. Используй тестовую карту ЮKassa: 2200 7000 0000 0000, срок 12/30, CVC 123
  4. Проверь логи сервера: должна появиться запись [YOOKASSA] Токен карты ****0000 сохранен...
  5. Проверь коллекцию cards в MongoDB: должна быть новая запись с cardToken
  6. Вызови POST /api/getMarkInfo с валидным кодом марки
  7. Проверь логи: должна быть запись [YOOKASSA] Успешное списание 1 RUB...
  8. Убедись, что в ответе /api/getMarkInfo есть paymentStatus: 'charged'
