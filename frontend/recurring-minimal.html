Рекуррентные платежи: минимум кода

Третья инструкция. Только суть. Повтори, чтобы закрепить.

1. Что нужно

  • Node.js (последняя версия)
  • MongoDB (локально или MongoDB Atlas)
  • Аккаунт ЮKassa с Shop ID и Secret Key

2. Создать проект

Открой терминал в VS Code (Ctrl + `) и выполни:

mkdir recurring-demo
cd recurring-demo
npm init -y
npm install express mongoose axios dotenv

3. Создать файлы

Создай три файла в корне проекта:

  • .env — настройки
  • server.js — весь код
  • index.html — одна кнопка

4. Файл .env

PORT=3000
MONGODB_URI=mongodb://localhost:27017/recurring-demo
YOOKASSA_SHOP_ID=твой_shop_id
YOOKASSA_SECRET_KEY=твой_secret_key
FRONTEND_URL=http://localhost:3000

5. Файл server.js

Весь код в одном файле. Четыре блока.

import dotenv from 'dotenv';
dotenv.config();

import express from 'express';
import mongoose from 'mongoose';
import axios from 'axios';
import path from 'path';
import { fileURLToPath } from 'url';

const app = express();
app.use(express.json());
app.use(express.static('public'));

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// Подключение к MongoDB
await mongoose.connect(process.env.MONGODB_URI);
console.log('MongoDB подключен');

// Модель карты
const CardSchema = new mongoose.Schema({
  userId: String,
  cardToken: String,
  last4: String
});
const Card = mongoose.model('Card', CardSchema);

// 1. Привязка карты
app.post('/api/save-card', async (req, res) => {
  const { userId } = req.body;

  const response = await axios.post(
    'https://api.yookassa.ru/v3/payments',
    {
      amount: { value: '1.00', currency: 'RUB' },
      capture: true,
      save_payment_method: true,
      payment_method_data: { type: 'bank_card' },
      metadata: { userId },
      confirmation: {
        type: 'redirect',
        return_url: `${process.env.FRONTEND_URL}/success.html`
      }
    },
    {
      auth: {
        username: process.env.YOOKASSA_SHOP_ID,
        password: process.env.YOOKASSA_SECRET_KEY
      }
    }
  );

  res.json({ url: response.data.confirmation.confirmation_url });
});

// 2. Вебхук от ЮKassa
app.post('/api/webhook', async (req, res) => {
  const event = req.body;

  if (event.event === 'payment.succeeded') {
    const payment = event.object;

    if (payment.payment_method?.saved && payment.amount.value === '1.00') {
      const userId = payment.metadata.userId;
      const cardToken = payment.payment_method.id;
      const last4 = payment.payment_method.card.last4;

      await Card.create({ userId, cardToken, last4 });
      console.log('Карта сохранена:', last4);
    }
  }

  res.status(200).json({ ok: true });
});

// 3. Списание 1 рубля
app.post('/api/charge', async (req, res) => {
  const { userId } = req.body;

  const card = await Card.findOne({ userId });
  if (!card) return res.json({ error: 'Карта не найдена' });

  const response = await axios.post(
    'https://api.yookassa.ru/v3/payments',
    {
      amount: { value: '1.00', currency: 'RUB' },
      capture: true,
      description: 'Тестовое списание',
      payment_method_id: card.cardToken
    },
    {
      auth: {
        username: process.env.YOOKASSA_SHOP_ID,
        password: process.env.YOOKASSA_SECRET_KEY
      }
    }
  );

  res.json({ status: response.data.status });
});

app.listen(process.env.PORT, () => {
  console.log(`Сервер запущен на порту ${process.env.PORT}`);
});

В package.json добавь:

"type": "module"

6. Файл public/index.html

Создай папку public и файл index.html внутри:

<!DOCTYPE html>
<html>
<head><title>Тест</title></head>
<body>
  <h1>Рекуррентные платежи</h1>
  <button onclick="saveCard()">Привязать карту</button>
  <button onclick="charge()">Списать 1 рубль</button>

  <script>
    const userId = 'user-123';

    async function saveCard() {
      const res = await fetch('/api/save-card', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ userId })
      });
      const data = await res.json();
      window.location.href = data.url;
    }

    async function charge() {
      const res = await fetch('/api/charge', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ userId })
      });
      const data = await res.json();
      alert('Статус: ' + data.status);
    }
  </script>
</body>
</html>

7. Запустить локально

node server.js

Открой http://localhost:3000

8. Настроить вебхук

Для тестов используй ngrok:

ngrok http 3000

Скопируй URL из ngrok (например, https://abc123.ngrok.io) и добавь в ЮKassa:

  • Настройки → HTTP-уведомления
  • URL: https://abc123.ngrok.io/api/webhook
  • Событие: payment.succeeded

9. Как это работает

  1. Нажимаешь "Привязать карту" → редирект на ЮKassa
  2. Вводишь тестовую карту: 2200 7000 0000 0000, срок 12/30, CVC 123
  3. ЮKassa шлёт вебхук → сервер сохраняет токен в MongoDB
  4. Нажимаешь "Списать 1 рубль" → сервер берёт токен из БД и списывает деньги

10. Схема

Кнопка "Привязать" → ЮKassa → Вебхук → Сохранить токен в БД
                                                              ↓
Кнопка "Списать" → Взять токен из БД → ЮKassa → Деньги списаны

Всё. Больше ничего не нужно.
